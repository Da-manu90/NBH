// file: notify_partner_assigned.ts

import { serve } from "https://deno.land/std@0.131.0/http/server.ts";
import { createClient } from "https://esm.sh/@supabase/supabase-js";

// Supabase-Client:
const supabase = createClient(
  Deno.env.get("https://entpyczncbnavvudfgrb.supabase.co")!,
  Deno.env.get("SUPABASE SECRET KEY")!
);

// OneSignal-Konstanten
const ONESIGNAL_APP_ID = "6f7988be-911b-4802-80d6-73154afbb487";
const ONESIGNAL_REST_API_KEY = "os_v2_app_ONESIG REST API KEY";

// Du könntest den Secret-Token auch aus einer ENV-Var holen:
const REQUIRED_BEARER_TOKEN = "MEIN_SUPER_SECRET_TOKEN";

serve(async (req) => {
  try {
    // --- Auth-Check-Block -----------------------------------------
    const authHeader = req.headers.get("Authorization");
    if (!authHeader || !authHeader.startsWith("Bearer ")) {
      return new Response(JSON.stringify({ error: "Unauthorized" }), { status: 401 });
    }
    const token = authHeader.substring("Bearer ".length);
    if (token !== REQUIRED_BEARER_TOKEN) {
      return new Response(JSON.stringify({ error: "Invalid token" }), { status: 403 });
    }
    // ---------------------------------------------------------------

    // parse body => { userId, partnerId, entryId }
    const { userId, partnerId, entryId } = await req.json();

    // 1) PartnerName ermitteln:
    const { data: partnerUser, error: partnerError } = await supabase
      .from("users")
      .select("username")
      .eq("id", partnerId)
      .single();
    if (partnerError) throw partnerError;

    const partnerName = partnerUser?.username ?? "Ein Partner";

    // 2) Ersteller-PlayerId aus DB
    const { data: user, error: userError } = await supabase
      .from("users")
      .select("player_id")
      .eq("id", userId)
      .single();
    if (userError) throw userError;

    if (!user?.player_id) {
      return new Response("No player_id for user => can't send push", { status: 200 });
    }

    // 3) OneSignal-Request
    const body = {
      app_id: ONESIGNAL_APP_ID,
      include_player_ids: [user.player_id],
      headings: { "en": `${partnerName} hat Ihr Event angenommen.` },
      contents: { "en": "Bestätigen Sie jetzt, um das Event zu starten." },
      data: {
        entryId: entryId
      },
      url: `nbh://detailpage?entryId=${entryId}`
    };

    const resp = await fetch("https://onesignal.com/api/v1/notifications", {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        "Authorization": `Basic ${ONESIGNAL_REST_API_KEY}`
      },
      body: JSON.stringify(body)
    });

    if (!resp.ok) {
      const errorText = await resp.text();
      return new Response(`OneSignal error: ${errorText}`, { status: 400 });
    }

    return new Response("Push sent successfully!", { status: 200 });

  } catch (err) {
    return new Response(`Edge Function error => ${err.message}`, { status: 500 });
  }
});
